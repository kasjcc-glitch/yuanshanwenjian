/** * 评论展开/收起功能 - 独立JS文件 * * 使用方法： * 1. 将本文件保存为 comments.js * 2. 在HTML页面中引入： * 3. 在按钮上添加 onclick="toggleComments(this)" * 4. 确保评论区域(div.comment-box)紧跟在按钮(button.comment-toggle)后面 * * 按钮文字格式：[ 展开 N 条评论 ] 或 [ 收起 N 条评论 ] */ function toggleComments(btn) { const box = btn.nextElementSibling; const isOpen = box.classList.contains('open'); if (isOpen) { box.classList.remove('open'); // 恢复"展开"文字，保留评论数 const match = btn.innerText.match(/\d+/); if (match) { btn.innerText = '[ 展开 ' + match[0] + ' 条评论 ]'; } } else { box.classList.add('open'); btn.innerText = btn.innerText.replace('展开', '收起'); } }
